CREATE TABLE IF NOT EXISTS manufacturer(
    	ID                          		INTEGER PRIMARY KEY AUTOINCREMENT,
    	NAME                        		TEXT NOT NULL,
   	CATEGORY                   		INTEGER CHECK (CATEGORY IN (0, 1, 2)) DEFAULT 0
);
                     
CREATE TABLE IF NOT EXISTS inverter (
	ID                          		INTEGER PRIMARY KEY AUTOINCREMENT,
	MODEL                       		TEXT NOT NULL,
	MANUFACTURER_ID             		INTEGER NOT NULL,
-- Dimensões e Especificações
	DIM_WIDTH                   		REAL DEFAULT -1,
	DIM_HEIGHT                  		REAL DEFAULT -1,
	DIM_DEPTH                   		REAL DEFAULT -1,
	DIM_WEIGHT                  		REAL DEFAULT -1,
	MAX_OPERATING_TEMPERATURE		INTEGER,
	MIN_OPERATING_TEMPERATURE		INTEGER,
	COOLING_MODE				TEXT NOT NULL,
	PROTECTION_DEGREE			TEXT NOT NULL,
	TOPOLOGY				TEXT NOT NULL,
-- Saída
	RATED_ACTIVE_POWER			INTEGER DEFAULT -1,
	MAX_ACTIVE_POWER			INTEGER DEFAULT -1,
	RATED_OUTPUT_VOLTAGE			REAL DEFAULT -1,
	RATED_OUTPUT_CURRENT			REAL DEFAULT -1,
	MAX_OUTPUT_CURRENT			REAL DEFAULT -1,
-- Entrada
    	OVERLOAD                    		INTEGER DEFAULT -1,
	NUMBER_OF_TRACKERS			INTEGER DEFAULT -1,
	NUMBER_OF_INPUTS			INTEGER DEFAULT -1,
-- Status
	ACTIVE					INTEGER NOT NULL CHECK (ACTIVE IN (0, 1)),
               
    FOREIGN KEY (MANUFACTURER_ID) REFERENCES manufacturer(ID)
);

CREATE TABLE IF NOT EXISTS mppt(
	ID				        INTEGER PRIMARY KEY AUTOINCREMENT,
    	INVERTER_ID                 		INTEGER NOT NULL,
	MPPT_INDEX			        INTEGER DEFAULT 0,
	NUMBER_OF_INPUTS		   	INTEGER DEFAULT -1,
	MAX_INPUT_VOLTAGE		    	INTEGER DEFAULT -1,
	MIN_STARTUP_VOLTAGE		    	INTEGER DEFAULT -1,
	MAX_OPERATING_VOLTAGE			INTEGER DEFAULT -1,
	MIN_OPERATING_VOLTAGE			INTEGER DEFAULT -1,
	MAX_FULL_LOAD_VOLTAGE			INTEGER DEFAULT -1,
	MIN_FULL_LOAD_VOLTAGE			INTEGER DEFAULT -1,
	RATED_INPUT_VOLTAGE		    	INTEGER DEFAULT -1,
	MAX_SHORT_CIRCUIT_CURRENT		REAL DEFAULT -1,
	MAX_OPERATING_CURRENT			REAL DEFAULT -1,
	FOREIGN KEY (INVERTER_ID) REFERENCES inverter(ID) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS inverter_system (
	ID					INTEGER PRIMARY KEY AUTOINCREMENT,
	INVERTER_ID				INTEGER NOT NULL,
	SYSTEM_TYPE				TEXT CHECK(SYSTEM_TYPE IN ('ON-GRID', 'OFF-GRID', 'GRIDZERO', 'HYBRID')),
	FOREIGN KEY (INVERTER_ID) REFERENCES inverter(ID) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS inverter_communication (
	ID 				 	INTEGER PRIMARY KEY AUTOINCREMENT,
	INVERTER_ID			        INTEGER NOT NULL,
	COMMUNICATION_TYPE		    	TEXT CHECK(COMMUNICATION_TYPE IN ('DISPLAY', 'RS485', 'WIFI', 'LED', 'USB')),
	FOREIGN KEY (INVERTER_ID) REFERENCES inverter(ID) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS inverter_output_mode(
	ID				        INTEGER PRIMARY KEY AUTOINCREMENT,
	INVERTER_ID			        INTEGER NOT NULL,
	OUTPUT_MODE			        TEXT CHECK(OUTPUT_MODE IN ('THREE_PHASE_FOUR_WIRE', 'THREE_PHASE_THREE_WIRE', 'SINGLE_PHASE')),
	FOREIGN KEY (INVERTER_ID) REFERENCES inverter(ID) ON DELETE CASCADE
);
           
CREATE TABLE IF NOT EXISTS module (
	ID					INTEGER PRIMARY KEY AUTOINCREMENT,
	MODEL					TEXT NOT NULL,
	MANUFACTURER_ID				INTEGER NOT NULL,
-- Dimensões e especificações
	DIM_WIDTH				REAL DEFAULT -1,
	DIM_HEIGHT				REAL DEFAULT -1,
	DIM_DEPTH				REAL DEFAULT -1,
	DIM_WEIGHT				REAL DEFAULT -1,
-- Dados elétricos (STC)
	WP					INTEGER DEFAULT -1,
	VMPP					REAL DEFAULT -1,
	IMPP					REAL DEFAULT -1,
	VOC					REAL DEFAULT -1,
	ISC					REAL DEFAULT -1,
-- Dados Mecânicos
	SOLAR_CELLS				TEXT CHECK (SOLAR_CELLS IN ('MONOCRISTALINO', 'POLICRISTALINO')),
	CELL_TYPE				TEXT CHECK (CELL_TYPE IN ('FULL CELL', 'HALF CELL')),
	SURFACE_TYPE				TEXT CHECK (SURFACE_TYPE IN ('MONOFACIAL', 'BIFACIAL')),
-- Coeficientes de temperatura
	COEF_PMAX				REAL DEFAULT -0.35,
	COEF_VOC				REAL DEFAULT -0.3,
	COEF_ISC				REAL DEFAULT 0.05,
-- Status
	ACTIVE					INTEGER NOT NULL CHECK (ACTIVE IN (0, 1)),

	FOREIGN KEY (MANUFACTURER_ID) REFERENCES manufacturer(ID) 
);
    
-- Índices para otimizar buscas
CREATE INDEX IF NOT EXISTS idx_mppt_inverter_id ON mppt (INVERTER_ID);
CREATE INDEX IF NOT EXISTS idx_system_inverter_id ON inverter_system (INVERTER_ID);
CREATE INDEX IF NOT EXISTS idx_comm_inverter_id ON inverter_communication (INVERTER_ID);
CREATE INDEX IF NOT EXISTS idx_output_inverter_id ON inverter_output_mode (INVERTER_ID);
CREATE INDEX IF NOT EXISTS idx_module_manufacturer_id ON module (MANUFACTURER_ID);
CREATE INDEX IF NOT EXISTS idx_inverter_manufacturer_id ON inverter (MANUFACTURER_ID);
CREATE INDEX IF NOT EXISTS idx_inverter_model ON inverter (MODEL);
CREATE INDEX IF NOT EXISTS idx_module_model ON module (MODEL);
